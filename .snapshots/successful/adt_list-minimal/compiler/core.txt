CoreProgram
    { cdecls =
        [ CFunDef "show" [ "xs" ]
            ( CCall
                ( CVar "$cps$show" )
                [ CVar "xs"
                , CCon 0 []
                ]
            )
        , CFunDef "$cps$show"
            [ "xs"
            , "$k"
            ]
            ( CLoop
                ( CCase
                    ( CVar "xs" )
                    [
                        ( 0
                        ,
                            [ "h"
                            , "t"
                            ]
                        , CContinue
                            [ CVar "t"
                            , CCon 1
                                [ CVar "$k"
                                , CVar "h"
                                ]
                            ]
                        )
                    ,
                        ( 1
                        , []
                        , CCall
                            ( CVar "$apply$show" )
                            [ CVar "$k"
                            , CString ""
                            ]
                        )
                    ]
                )
            )
        , CFunDef "$apply$show"
            [ "$k"
            , "$x"
            ]
            ( CLoop
                ( CCase
                    ( CVar "$k" )
                    [
                        ( 0
                        , []
                        , CVar "$x"
                        )
                    ,
                        ( 1
                        ,
                            [ "$pk_1"
                            , "h"
                            ]
                        , CContinue
                            [ CVar "$pk_1"
                            , CCall
                                ( CBuiltIn "concatString" )
                                [ CCall
                                    ( CBuiltIn "concatString" )
                                    [ CVar "h"
                                    , CString ","
                                    ]
                                , CVar "$x"
                                ]
                            ]
                        )
                    ]
                )
            )
        , CFunDef "main" [ "_input" ]
            ( CCall
                ( CBuiltIn "IO.Stdout.print" )
                [ CCall
                    ( CVar "show" )
                    [ CCon 0
                        [ CString "a"
                        , CCon 0
                            [ CString "b"
                            , CCon 0
                                [ CString "c"
                                , CCon 1 []
                                ]
                            ]
                        ]
                    ]
                ]
            )
        ]
    }
