CoreProgram
    { cdecls =
        [ CFunDef "show" [ "xs" ]
            ( CCall
                ( CVar "$cps$show" )
                [ CVar "xs"
                , CCon 0 []
                ]
            )
        , CFunDef "$cps$show"
            [ "xs"
            , "$k"
            ]
            ( CLoop
                ( CCase
                    ( CVar "xs" )
                    [
                        ( 0
                        ,
                            [ "h"
                            , "t"
                            ]
                        , CContinue
                            [ CVar "t"
                            , CCon 1
                                [ CVar "$k"
                                , CVar "h"
                                ]
                            ]
                        )
                    ,
                        ( 1
                        , []
                        , CCall
                            ( CVar "$apply$show" )
                            [ CVar "$k"
                            , CString ""
                            ]
                        )
                    ]
                )
            )
        , CFunDef "$apply$show"
            [ "$k"
            , "$x"
            ]
            ( CLoop
                ( CCase
                    ( CVar "$k" )
                    [
                        ( 0
                        , []
                        , CVar "$x"
                        )
                    ,
                        ( 1
                        ,
                            [ "$pk_1"
                            , "h"
                            ]
                        , CContinue
                            [ CVar "$pk_1"
                            , CCall
                                ( CBuiltIn "concatString" )
                                [ CCall
                                    ( CBuiltIn "concatString" )
                                    [ CVar "h"
                                    , CString ","
                                    ]
                                , CVar "$x"
                                ]
                            ]
                        )
                    ]
                )
            )
        , CFunDef "shout" [ "s" ]
            ( CCall
                ( CBuiltIn "concatString" )
                [ CVar "s"
                , CString "!"
                ]
            )
        , CFunDef "main" [ "_input" ]
            ( CCall
                ( CBuiltIn "IO.Stdout.print" )
                [ CCall
                    ( CVar "show" )
                    [ CCall
                        ( CVar "$df$map$0" )
                        [ CCon 0
                            [ CString "a"
                            , CCon 0
                                [ CString "b"
                                , CCon 0
                                    [ CString "c"
                                    , CCon 1 []
                                    ]
                                ]
                            ]
                        ]
                    ]
                ]
            )
        , CFunDef "$df$map$0" [ "list" ]
            ( CCall
                ( CVar "$cps$$df$map$0" )
                [ CVar "list"
                , CCon 0 []
                ]
            )
        , CFunDef "$cps$$df$map$0"
            [ "list"
            , "$k"
            ]
            ( CLoop
                ( CCase
                    ( CVar "list" )
                    [
                        ( 0
                        ,
                            [ "head"
                            , "tail"
                            ]
                        , CContinue
                            [ CVar "tail"
                            , CCon 1
                                [ CVar "$k"
                                , CVar "head"
                                ]
                            ]
                        )
                    ,
                        ( 1
                        , []
                        , CCall
                            ( CVar "$apply$$df$map$0" )
                            [ CVar "$k"
                            , CCon 1 []
                            ]
                        )
                    ]
                )
            )
        , CFunDef "$apply$$df$map$0"
            [ "$k"
            , "$x"
            ]
            ( CLoop
                ( CCase
                    ( CVar "$k" )
                    [
                        ( 0
                        , []
                        , CVar "$x"
                        )
                    ,
                        ( 1
                        ,
                            [ "$pk_1"
                            , "head"
                            ]
                        , CContinue
                            [ CVar "$pk_1"
                            , CCon 0
                                [ CCall
                                    ( CVar "shout" )
                                    [ CVar "head" ]
                                , CVar "$x"
                                ]
                            ]
                        )
                    ]
                )
            )
        ]
    }
