            ( CCase
                ( CVar "pair" )
                [
                    ( 0
                    ,
                        [ "first"
                        , "second"
                        ]
                    , CCall ( CPrim PrimConcat )
                        [ CCall ( CPrim PrimConcat )
                            [ CCall ( CPrim PrimConcat )
                                [ CCall ( CPrim PrimConcat )
                                    [ CString "("
                                    , CVar "first"
                                    ]
                                , CString ", "
                                ]
                            , CVar "second"
                            ]
                        , CString ")"
                        ]
                    )
                ]
            )
        , CFunDef "main" [ "input" ]
            ( CCall ( CPrim PrimPrint )
                [ CCall
                    ( CVar "showPair" )
                    [ CCon 0
                        [ CString "hello"
                        , CString "world"
                        ]
                    ]
                ]
            )
        , CFunDef "$con$Pair"
            [ "$x0"
            , "$x1"
            ]
            ( CCon 0
                [ CVar "$x0"
                , CVar "$x1"
                ]
            )
        ]
